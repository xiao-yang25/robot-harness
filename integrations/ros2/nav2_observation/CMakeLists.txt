cmake_minimum_required(VERSION 3.16)
project(robot_harness_nav2_observation LANGUAGES CXX)

# An explicit, source-tree-only build; the normal Core build has no ROS dependency.
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(lifecycle_msgs REQUIRED)
include(CTest)
add_subdirectory(../../.. core EXCLUDE_FROM_ALL)
add_executable(robot_harness_nav2_observation navigation_observation.cpp)
target_compile_features(robot_harness_nav2_observation PRIVATE cxx_std_17)
target_compile_options(robot_harness_nav2_observation PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(robot_harness_nav2_observation robot_harness_core)
ament_target_dependencies(robot_harness_nav2_observation rclcpp rclcpp_action nav2_msgs
  nav_msgs geometry_msgs rosgraph_msgs lifecycle_msgs)

if(BUILD_TESTING)
  add_executable(nav2_motion_window_test motion_window_test.cpp)
  target_compile_features(nav2_motion_window_test PRIVATE cxx_std_17)
  add_test(NAME nav2_motion_window COMMAND nav2_motion_window_test)
  add_test(NAME nav2_requires_fresh_deployment
    COMMAND ${CMAKE_COMMAND}
      -DNAV2_BINARY=$<TARGET_FILE:robot_harness_nav2_observation>
      -P ${CMAKE_CURRENT_SOURCE_DIR}/deployment_guard_test.cmake)
endif()

option(ROBOT_HARNESS_NAV2_SETTLEMENT "Build the isolated native settlement example" OFF)
if(ROBOT_HARNESS_NAV2_SETTLEMENT)
  find_package(nlohmann_json 3.10 REQUIRED)
  find_package(std_msgs REQUIRED)
  find_package(action_msgs REQUIRED)
  find_package(unique_identifier_msgs REQUIRED)
  find_package(std_srvs REQUIRED)
  find_package(m4_drive_probe REQUIRED)
  add_executable(robot_harness_nav2_settlement navigation_observation.cpp native_closure.cpp)
  target_compile_features(robot_harness_nav2_settlement PRIVATE cxx_std_17)
  target_compile_options(robot_harness_nav2_settlement PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_definitions(robot_harness_nav2_settlement PRIVATE ROBOT_HARNESS_NAV2_SETTLEMENT)
  target_link_libraries(robot_harness_nav2_settlement robot_harness_core nlohmann_json::nlohmann_json)
  ament_target_dependencies(robot_harness_nav2_settlement rclcpp rclcpp_action nav2_msgs
    nav_msgs geometry_msgs rosgraph_msgs lifecycle_msgs std_msgs std_srvs m4_drive_probe action_msgs unique_identifier_msgs)
  if(BUILD_TESTING)
    add_executable(nav2_closure_observations_test closure_observations_test.cpp)
    target_compile_features(nav2_closure_observations_test PRIVATE cxx_std_17)
    target_link_libraries(nav2_closure_observations_test nlohmann_json::nlohmann_json)
    add_test(NAME nav2_closure_observations COMMAND nav2_closure_observations_test)
    add_executable(nav2_runtime_observation_watch_test runtime_observation_watch_test.cpp)
    target_compile_features(nav2_runtime_observation_watch_test PRIVATE cxx_std_17)
    add_test(NAME nav2_runtime_observation_watch COMMAND nav2_runtime_observation_watch_test)
    add_executable(nav2_cancel_response_test cancel_response_test.cpp)
    target_compile_features(nav2_cancel_response_test PRIVATE cxx_std_17)
    target_link_libraries(nav2_cancel_response_test robot_harness_core)
    ament_target_dependencies(nav2_cancel_response_test action_msgs unique_identifier_msgs)
    add_test(NAME nav2_cancel_response COMMAND nav2_cancel_response_test)
  endif()
endif()
