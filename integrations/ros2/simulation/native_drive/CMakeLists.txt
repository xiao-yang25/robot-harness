cmake_minimum_required(VERSION 3.16)
project(robot_harness_simulation_drive LANGUAGES C CXX)
find_package(ament_cmake REQUIRED)
find_package(m4_drive_probe REQUIRED)
find_package(gazebo REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(gazebo_plugins 3.9.0 EXACT REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
add_library(scoped_diff_drive SHARED scoped_diff_drive.cpp)
target_compile_features(scoped_diff_drive PRIVATE cxx_std_17)
target_include_directories(scoped_diff_drive PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_libraries(scoped_diff_drive ${GAZEBO_LIBRARIES})
ament_target_dependencies(scoped_diff_drive m4_drive_probe gazebo_ros gazebo_plugins
  geometry_msgs nav_msgs std_msgs tf2_ros tf2_geometry_msgs)
install(TARGETS scoped_diff_drive LIBRARY DESTINATION lib)
