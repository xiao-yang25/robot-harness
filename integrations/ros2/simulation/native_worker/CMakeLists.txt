cmake_minimum_required(VERSION 3.16)
project(m4_native_worker_probe LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(nav2_controller 1.1.20 EXACT REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
add_executable(native_worker_controller native_worker_controller.cpp)
target_compile_features(native_worker_controller PRIVATE cxx_std_17)
target_compile_options(native_worker_controller PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(native_worker_controller nav2_controller rclcpp std_msgs std_srvs)

find_package(nav2_behavior_tree 1.1.20 EXACT REQUIRED)
add_library(scoped_follow_path SHARED scoped_follow_path.cpp)
target_compile_features(scoped_follow_path PRIVATE cxx_std_17)
target_compile_definitions(scoped_follow_path PRIVATE BT_PLUGIN_EXPORT)
target_compile_options(scoped_follow_path PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(scoped_follow_path nav2_behavior_tree rclcpp std_msgs)

find_package(nav2_planner 1.1.20 EXACT REQUIRED)
find_package(nav2_bt_navigator 1.1.20 EXACT REQUIRED)
add_executable(native_worker_planner native_worker_planner.cpp)
target_compile_features(native_worker_planner PRIVATE cxx_std_17)
target_compile_options(native_worker_planner PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(native_worker_planner nav2_planner rclcpp std_msgs std_srvs)
add_executable(native_scope_navigator native_scope_navigator.cpp)
target_compile_features(native_scope_navigator PRIVATE cxx_std_17)
target_compile_options(native_scope_navigator PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(native_scope_navigator nav2_bt_navigator rclcpp std_srvs)
add_library(scoped_compute_path SHARED scoped_compute_path.cpp)
target_compile_features(scoped_compute_path PRIVATE cxx_std_17)
target_compile_definitions(scoped_compute_path PRIVATE BT_PLUGIN_EXPORT)
target_compile_options(scoped_compute_path PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(scoped_compute_path nav2_behavior_tree rclcpp std_msgs)

include(CTest)
if(BUILD_TESTING)
  add_executable(tree_halt_test tree_halt_test.cpp)
  target_compile_features(tree_halt_test PRIVATE cxx_std_17)
  ament_target_dependencies(tree_halt_test nav2_behavior_tree)
  add_test(NAME native_tree_halt COMMAND tree_halt_test)
endif()
